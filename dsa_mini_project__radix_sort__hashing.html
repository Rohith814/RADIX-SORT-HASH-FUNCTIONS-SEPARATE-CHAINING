Project Overview

Radix Sort

Radix Sort is a non-comparison-based sorting algorithm. This project uses LSD radix sort, where numbers are sorted digit by digit from least significant digit to most significant digit.

Hashing with Separate Chaining

Hashing maps a key to an index using a hash function. If multiple keys map to the same index, separate chaining stores them in a linked-list-like chain.

Radix Sort in C Style

#include <stdio.h>

int getMax(int arr[], int n) {
    int max = arr[0];

for (int i = 1; i < n; i++) {
        if (arr[i] > max) {
            max = arr[i];
        }
    }

return max;
}

void countingSort(int arr[], int n, int exp) {
    int output[n];
    int count[10] = {0};

for (int i = 0; i < n; i++) {
        count[(arr[i] / exp) % 10]++;
    }

for (int i = 1; i < 10; i++) {
        count[i] += count[i - 1];
    }

for (int i = n - 1; i >= 0; i--) {
        int digit = (arr[i] / exp) % 10;
        output[count[digit] - 1] = arr[i];
        count[digit]--;
    }

for (int i = 0; i < n; i++) {
        arr[i] = output[i];
    }
}

void radixSort(int arr[], int n) {
    int max = getMax(arr, n);

for (int exp = 1; max / exp > 0; exp *= 10) {
        countingSort(arr, n, exp);
    }
}

Hashing with Separate Chaining in C Style

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node* next;
};

int hashFunction(int key, int size) {
    return key % size;
}

void insert(struct Node* table[], int size, int key) {
    int index = hashFunction(key, size);

struct Node* newNode = (struct Node*)malloc(sizeof(struct Node));
    newNode->data = key;
    newNode->next = table[index];

table[index] = newNode;
}

int search(struct Node* table[], int size, int key) {
    int index = hashFunction(key, size);
    struct Node* temp = table[index];

while (temp != NULL) {
        if (temp->data == key) {
            return 1;
        }
        temp = temp->next;
    }

return 0;
}

void deleteKey(struct Node* table[], int size, int key) {
    int index = hashFunction(key, size);
    struct Node* temp = table[index];
    struct Node* prev = NULL;

while (temp != NULL && temp->data != key) {
        prev = temp;
        temp = temp->next;
    }

if (temp == NULL) {
        return;
    }

if (prev == NULL) {
        table[index] = temp->next;
    } else {
        prev->next = temp->next;
    }

free(temp);
}
